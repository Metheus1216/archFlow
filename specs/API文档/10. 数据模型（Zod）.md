> 四个约定，评审时请确认：
> 
> 1. `.nullable()` = **字段必返、值可为 ****`null`**；`.optional()` = **字段可能不存在**（§0\.1 \#28）。两者不可混用。
> 
> 2. 响应模型只校验「结构 \+ 显式空值 \+ 枚举」，不校验业务长度；长度、格式等约束属于请求模型。

>
> 4. **字段长度 / 格式等约束以本文件为唯一源**；API 文档只描述语义，不重复数值。若两者冲突，以本文件为准（§1\.1）。
> 
> 



**分类沿用原《Zod Schema》的文件划分**：基础类型、枚举、图与通用包装放 `common.schema.ts`；`auth` / `project` / `agent` 各自成文件，`knowledge`、`asset`、`conversation` 收在 `project` 之下。原文档没有的 `Run`、`RunError` 按语义并入 `agent.schema.ts`。

```TypeScript
/**
 * 架构图 AI 平台 · API 数据模型（Zod 4）
 * 权威源：API 文档 v1.2 §10。所有 TypeScript 类型由 z.infer 派生，不再手写 interface。
 * 文件划分沿用《Zod Schema》：common / auth / project（含 knowledge、asset、conversation）/ agent。
 */
import { z } from "zod";

/* ───────────────────────── 基础类型 ───────────────────────── */

/** 资源主键（§1.1）：不透明字符串，前端不得解析、不得当数字处理。 */
export const Id = z
  .string()
  .min(1)
  .max(64)
  .regex(/^[A-Za-z0-9_-]+$/, "id 只允许字母、数字、下划线、连字符")
  .describe("资源主键（不透明字符串；对外 id 即库内 uuid 列，不做格式强校验）");

/** 时间（§1.1）：ISO-8601 带时区偏移，服务端统一输出 +08:00。 */
export const ISODateTime = z.iso
  .datetime({ offset: true })
  .describe("ISO-8601 时间，带时区偏移");

/**
 * 链路 ID（§1.8）：服务端生成 16 位十六进制；
 * 但客户端传入合法 X-Request-Id 时会原样沿用，
 * 因此不能假定它一定是 16 位 hex，按 [A-Za-z0-9-]{8,64} 校验。
 */
export const TraceId = z
  .string()
  .regex(/^[A-Za-z0-9-]{8,64}$/, "traceId 只允许 [A-Za-z0-9-]，长度 8–64")
  .describe("链路追踪 ID");

/** 非负整数：计数、token 数、字节数、延迟。 */
export const Count = z.number().int().nonnegative();

/* ───────────────────────── 枚举（§1.10） ───────────────────────── */

/** system 仅后端使用；GET /messages 响应会过滤，不下发前端（§10.2）。 */
export const MessageRole = z.enum(["user", "assistant", "system"]);
export type MessageRole = z.infer<typeof MessageRole>;

/** 文件类型：text/image 可解析；pdf/docx 仅支持上传，不解析（见 §7.1 能力矩阵）。 */
export const FileType = z.enum(["text", "image", "pdf", "docx"]);
export type FileType = z.infer<typeof FileType>;

/** stored = 已上传但按设计不解析（pdf/docx），不可被 Agent 引用。 */
export const FileStatus = z.enum(["processing", "ready", "failed", "stored"]);
export type FileStatus = z.infer<typeof FileStatus>;

export const CardKind = z.enum(["single", "multi", "text"]);
export type CardKind = z.infer<typeof CardKind>;

export const CardStatus = z.enum(["pending", "answered", "ignored", "expired"]);
export type CardStatus = z.infer<typeof CardStatus>;

export const RunStatus = z.enum(["queued", "running", "succeeded", "failed", "canceled"]);
export type RunStatus = z.infer<typeof RunStatus>;

export const FinishReason = z.enum(["stop", "length", "canceled", "error"]);
export type FinishReason = z.infer<typeof FinishReason>;

/* ───────────────────────── 图（§7.6 仅约束传输形态） ───────────────────────── */

export const GRAPH_MAX_BYTES = 512 * 1024; // 512KB，按 UTF-8 编码后的字节数
export const GRAPH_MAX_DEPTH = 6; // 嵌套深度，根对象记为第 1 层

/** 图数据：nodes / edges 内部结构 TBD，服务端视为不透明 JSON。 */
export const Graph = z.looseObject({
  nodes: z.array(z.unknown()),
  edges: z.array(z.unknown()),
  version: z.number().int().positive().optional(), // 兼容策略 TBD
});
export type Graph = z.infer<typeof Graph>;

const encoder = new TextEncoder();

function utf8ByteLength(value: unknown): number {
  return encoder.encode(JSON.stringify(value)).length;
}

/** 容器嵌套深度：对象/数组各算一层，原始值算 0 层。 */
function jsonDepth(value: unknown): number {
  if (Array.isArray(value)) {
    let max = 0;
    for (const item of value) max = Math.max(max, jsonDepth(item));
    return max + 1;
  }
  if (value !== null && typeof value === "object") {
    let max = 0;
    for (const item of Object.values(value as Record<string, unknown>)) {
      max = Math.max(max, jsonDepth(item));
    }
    return max + 1;
  }
  return 0;
}

/* 图相关错误（§1.5）：params 挂在 issue 上，供全局 Zod 异常过滤器直接映射。 */
export const GRAPH_TOO_LARGE_ERROR = { httpStatus: 413, errorCode: "GRAPH_TOO_LARGE" } as const;
export const GRAPH_INVALID_ERROR = { httpStatus: 422, errorCode: "GRAPH_INVALID" } as const;

/**
 * 入参图：在 Graph 之上叠加 §7.6 的大小与深度硬约束。
 * 只用于请求校验，响应解析用 Graph，避免对 512KB 图重复做序列化。
 *
 * 错误归属按原因、不按资源：
 * - 序列化超过 GRAPH_MAX_BYTES → 413 GRAPH_TOO_LARGE
 * - 结构 / 嵌套深度不合法（含 nodes、edges 不是数组）→ 422 GRAPH_INVALID
 */
export const GraphInput = Graph.superRefine((g, ctx) => {
  const bytes = utf8ByteLength(g);
  if (bytes > GRAPH_MAX_BYTES) {
    ctx.addIssue({
      code: "custom",
      message: `图数据 ${bytes} 字节，超过上限 ${GRAPH_MAX_BYTES} 字节（UTF-8）`,
      params: GRAPH_TOO_LARGE_ERROR,
    });
  }
  const depth = jsonDepth(g);
  if (depth > GRAPH_MAX_DEPTH) {
    ctx.addIssue({
      code: "custom",
      message: `图嵌套深度 ${depth} 层，超过上限 ${GRAPH_MAX_DEPTH} 层`,
      params: GRAPH_INVALID_ERROR,
    });
  }
});
export type GraphInput = z.infer<typeof GraphInput>;

type GraphError = { httpStatus: 413 | 422; errorCode: "GRAPH_TOO_LARGE" | "GRAPH_INVALID" };

/**
 * 把 GraphInput 抛出的 ZodError 映射成 API 错误：
 * 1. issue.params 带 errorCode（大小 / 深度）→ 用 issue 自带的状态与错误码；
 * 2. 其余结构错误（缺 nodes / edges、类型不对）→ 422 GRAPH_INVALID。
 * 全局 Zod 异常过滤器在 §4.3 / §8.1 / §8.4 等写图接口上调用本函数。
 */
export function mapGraphError(error: z.ZodError): GraphError {
  for (const issue of error.issues) {
    const params = (issue as { params?: Partial<GraphError> }).params;
    if (params?.errorCode) {
      return {
        httpStatus: params.httpStatus as GraphError["httpStatus"],
        errorCode: params.errorCode,
      };
    }
  }
  return { ...GRAPH_INVALID_ERROR };
}

/* ───────────────────────── 通用包装（§1.3 响应外壳 / 分页） ───────────────────────── */

/** 字段级报错，用于表单精准标红。 */
export const FieldError = z.object({
  field: z.string().min(1).describe("字段路径，嵌套用 .，数组用 [i]，如 cards[0].values"),
  code: z.string().regex(/^[A-Z][A-Z0-9_]*$/, "字段错误码为大写下划线"),
  message: z.string(),
});
export type FieldError = z.infer<typeof FieldError>;

/** 分页外壳：items + 游标，供前端无限滚动。 */
export const PageSchema = <T extends z.ZodType>(item: T) =>
  z.object({
    items: z.array(item),
    nextCursor: z.string().nullable(),
    hasMore: z.boolean(),
  });

/**
 * 泛型工厂没有具体的 T，`z.infer<typeof PageSchema>` 只能推出 `unknown`，
 * 因此类型别名手写；运行时校验仍由 `PageSchema` 完成（下同 `ApiResponse<T>`）。
 */
export type Page<T> = { items: T[]; nextCursor: string | null; hasMore: boolean };

/** 统一响应外壳：§1.3 失败时 data 为 null。 */
export const ApiResponseSchema = <T extends z.ZodType>(data: T) =>
  z.object({
    code: z.string().describe("大写英文常量，见 §1.5；成功为 OK"),  // v1.2：不再用数字编号
    message: z.string(),
    data: data.nullable(),
    errors: z.array(FieldError).optional(),
    traceId: TraceId,
  });

/** 同上：泛型工厂无法直接 `z.infer`，类型手写、校验走 `ApiResponseSchema`。 */
export type ApiResponse<T> = {
  code: string;
  message: string;
  data: T | null;
  errors?: FieldError[];
  traceId: string;
};

/* ───────────────────────── 请求参数（§1.3 分页） ───────────────────────── */

/** 分页查询参数：limit 默认 20、上限 50；cursor 为不透明游标。 */
export const PaginationQuery = z.object({
  limit: z.coerce.number().int().min(1).max(50).default(20),
  cursor: z.string().min(1).max(256).optional(),
});
export type PaginationQuery = z.infer<typeof PaginationQuery>;
```
